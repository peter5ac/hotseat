# Speed Tap Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Speed Tap game where players tap emojis in sequence order before a shrinking timer expires.

**Architecture:** `SpeedTapViewModel` drives all game state (round, sequence, timer, validation). `SpeedTapView` renders the UI over the shared camera+overlay stack. Integration into the existing app via `AppScreen` enum and `HomeView` game list.

**Tech Stack:** SwiftUI, `@Observable`, `Timer.publish` for countdown

## Global Constraints

- iOS 17+, Swift 5, SwiftUI only
- No persistent state (no UserDefaults, no files, no network)
- No scores/leaderboards — round count shown contextually only
- Large tap targets (pass-the-phone ready)
- All UI over camera preview + 0.4 opacity black overlay (handled by ContentView's existing ZStack)

- [ ] **Step 2: Implement SpeedTapViewModel**

```swift
@Observable
final class SpeedTapViewModel {
    var currentRound: Int = 1
    var sequence: [String] = []
    var tappedCount: Int = 0
    var grid: [String] = []
    var theme: EmojiTheme = emojiThemes[0]
    var timerProgress: Double = 1.0
    var gameState: SpeedTapState = .ready
    var showNice: Bool = false

    var roundsCompleted: Int { currentRound - 1 }

    private var timerDuration: Double {
        max(2.0, 5.0 - Double(currentRound - 1) * 0.3)
    }

    private var sequenceLength: Int {
        currentRound + 2
    }

    private var timer: Timer?

    var currentTarget: String? {
        guard tappedCount < sequence.count else { return nil }
        return sequence[tappedCount]
    }

    func startGame() {
        theme = emojiThemes.randomElement()!
        grid = theme.emojis
        currentRound = 1
        startRound()
    }

    func tap(_ emoji: String) {
        guard gameState == .playing else { return }
        guard let target = currentTarget else { return }

        if emoji == target {
            tappedCount += 1
            if tappedCount >= sequence.count {
                roundSuccess()
            }
        }
    }

    func playAgain() {
        currentRound = 1
        startRound()
    }

    func newRound() {
        gameState = .ready
        stopTimer()
    }

    private func startRound() {
        sequence = (0..<sequenceLength).map { _ in grid.randomElement()! }
        tappedCount = 0
        timerProgress = 1.0
        gameState = .playing
        startTimer()
    }

    private func roundSuccess() {
        stopTimer()
        gameState = .success
        showNice = true
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) { [weak self] in
            guard let self, self.gameState == .success else { return }
            self.showNice = false
            self.currentRound += 1
            self.startRound()
        }
    }

    private func startTimer() {
        stopTimer()
        let duration = timerDuration
        let interval: Double = 0.016 // ~60fps
        let decrement = interval / duration

        timer = Timer.scheduledTimer(withTimeInterval: interval, repeats: true) { [weak self] _ in
            guard let self else { return }
            DispatchQueue.main.async {
                self.timerProgress -= decrement
                if self.timerProgress <= 0 {
                    self.timerProgress = 0
                    self.timeUp()
                }
            }
        }
    }

    private func stopTimer() {
        timer?.invalidate()
        timer = nil
    }

    private func timeUp() {
        stopTimer()
        gameState = .gameOver
    }
}
```

```swift
import SwiftUI

struct SpeedTapView: View {
    @State private var viewModel = SpeedTapViewModel()
    let onExit: () -> Void
    @Environment(\.horizontalSizeClass) private var horizontalSizeClass

    private var isCompact: Bool {
        horizontalSizeClass == .compact
    }

    var body: some View {
        VStack(spacing: 0) {
            header
            Spacer()
            switch viewModel.gameState {
            case .ready:
                readyView
            case .playing, .success:
                gamePlayView
            case .gameOver:
                gameOverView
            }
            Spacer()
        }
        .onAppear {
            viewModel.startGame()
        }
    }
}
```
